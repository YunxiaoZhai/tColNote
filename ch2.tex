\section{Code}

\begin{codebox}{Fibonacci Numbers}{fib}
\begin{amzcode}{python}
def fib(n: int) -> int:
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a
\end{amzcode}
\end{codebox}
%
\begin{codebox}{Code from a File}{fib_file}
\amzinputcode{python}{attach/fib.py}
\end{codebox}

\section{Math Macros}
%
\begin{itemize}
    \item Paired delimiters: $\norm{\bs x}$, $\abs*{\frac{a}{b}}$, $\floor{x}$, $\cb{x\in\R:x>0}$.
    \item Restriction and derivatives: $\restrict{\pd{f}{x}}{x=0}$, $\odv{f}{x}$, $\pdv{f}{x,y}$.
    \item Optimization: $\max_{x}f(x)\st g(x)\le 0$.
\end{itemize}
